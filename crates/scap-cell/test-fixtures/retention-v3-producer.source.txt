//! Historical fixture producer. Build only in the unmodified b9f5128 worktree.
//! Uses the old public state machine and physical-image algorithm, never SQL.

use std::fmt::Write as _;
use std::fs;
use std::path::Path;

use raft::eraftpb::Entry;
use scap_cell::{
    decode_applied_outcome, encode_command, payload_digest, AdminResult, Admission,
    CellApplication, CellIdentity, CellState, Command, CommandKind, DatabaseIdentity, Grant,
    Limits, Outcome, Position, Reject, ADMIN, READ, WRITE,
};
use scap_core::{physical, CommitTicket};
use scap_raft_driver::Application;

const CELL: CellIdentity = CellIdentity {
    cell_id: 7,
    cell_incarnation: 8,
};
const DATABASE: DatabaseIdentity = DatabaseIdentity {
    database_id: 9,
    incarnation: 10,
    placement_generation: 11,
};
const GENESIS: Position = Position { term: 1, index: 1 };

fn limits() -> Limits {
    Limits {
        max_state_bytes: 65_536,
        max_command_bytes: 65_536,
        max_databases: 4,
        max_policy_grants: 4,
        max_admin_operations: 64,
        max_requests_per_database: 12,
        max_transaction_bytes: 8192,
        max_result_bytes: 1536,
    }
}

fn hex(bytes: &[u8]) -> String {
    let mut encoded = String::with_capacity(bytes.len() * 2);
    for byte in bytes {
        write!(&mut encoded, "{byte:02x}").unwrap();
    }
    encoded
}

fn rows(path: &Path, suffix: &str, values: &[Vec<u8>]) {
    let text: String = values
        .iter()
        .map(|value| format!("{}\n", hex(value)))
        .collect();
    fs::write(path.with_extension(suffix), text).unwrap();
}

fn outcome_name(outcome: &Outcome) -> String {
    match outcome {
        Outcome::Administrative(result) => match result {
            AdminResult::ArchivePublished(_) => "Administrative(ArchivePublished)".into(),
            AdminResult::ArchivePruned(_) => "Administrative(ArchivePruned)".into(),
            AdminResult::ArchiveHydrated(_) => "Administrative(ArchiveHydrated)".into(),
            other => format!("Administrative({other:?})"),
        },
        Outcome::Accepted(commit) => format!("Accepted({})", commit.receipt.sequence),
        Outcome::AlreadyCommitted(commit) => {
            format!("AlreadyCommitted({})", commit.receipt.sequence)
        }
        other => format!("{other:?}"),
    }
}

struct Trace {
    state: CellState,
    limits: Limits,
    image: physical::Image,
    commands: Vec<Vec<u8>>,
    outcomes: Vec<Vec<u8>>,
    prefixes: Vec<Vec<u8>>,
    labels: Vec<String>,
}

impl Trace {
    fn new(limits: Limits) -> Self {
        let state = CellState::bootstrap(CELL, 12, GENESIS, limits).unwrap();
        assert_eq!(state.physical_history_version(), 3);
        let prefixes = vec![state.encode()];
        let image = physical::Image::empty(
            physical::EngineIdentity {
                cell: CELL,
                database: DATABASE,
            },
            physical::Limits {
                max_database_pages: limits.max_state_bytes / physical::PAGE_SIZE as u32,
                max_changed_pages: limits.max_state_bytes / physical::PAGE_SIZE as u32,
                max_transaction_bytes: limits.max_transaction_bytes,
            },
        )
        .unwrap();
        let mut trace = Self {
            state, limits, image, commands: Vec::new(), outcomes: Vec::new(), prefixes,
            labels: vec!["row\tindex\tlabel\toutcome\trevision\tbase_revision\tcold\tstate_bytes\tstate_sha256".into()],
        };
        assert_eq!(
            trace.apply(
                "create",
                CommandKind::Create {
                    operation_id: 1,
                    policy: vec![Grant {
                        principal_id: 12,
                        permissions: READ | WRITE | ADMIN
                    }],
                }
            ),
            Outcome::Administrative(AdminResult::Created)
        );
        let claimed = trace.apply(
            "claim_owner",
            CommandKind::ClaimOwner {
                operation_id: 2,
                expected_epoch: 0,
                worker_instance_id: 15,
            },
        );
        let Outcome::Administrative(AdminResult::OwnerClaimed(token)) = claimed else {
            panic!("claim failed: {claimed:?}")
        };
        assert_eq!(
            trace.apply(
                "activate_owner",
                CommandKind::ActivateOwner {
                    operation_id: 3,
                    token,
                    restored_revision: 0,
                    restored_hash: [0; 32],
                }
            ),
            Outcome::Administrative(AdminResult::OwnerActivated)
        );
        trace
    }

    fn admission(&self) -> Admission {
        Admission {
            principal_id: 12,
            credential_id: 13,
            admission_id: 14,
            auth_epoch: self
                .state
                .database(DATABASE.database_id)
                .map_or(1, |db| db.auth_epoch),
        }
    }

    fn command(&self, kind: CommandKind) -> Command {
        Command {
            cell: CELL,
            database: DATABASE,
            admission: self.admission(),
            kind,
        }
    }

    fn apply(&mut self, label: &str, kind: CommandKind) -> Outcome {
        self.apply_command(label, self.command(kind))
    }

    fn apply_command(&mut self, label: &str, command: Command) -> Outcome {
        let bytes = encode_command(&command, self.limits).unwrap();
        assert_eq!(
            bytes[8], 2,
            "the old v3 implementation uses business envelope v2"
        );
        let position = Position {
            term: 1,
            index: self.state.position().index + 1,
        };
        let batch = CellApplication::new(CELL, self.limits)
            .unwrap()
            .apply_with_results(
                &self.state.encode(),
                &[Entry {
                    term: position.term,
                    index: position.index,
                    data: bytes.clone().into(),
                    ..Entry::default()
                }],
                &[position.index],
            )
            .unwrap();
        assert_eq!(batch.results.len(), 1);
        let outcome = decode_applied_outcome(&batch.results[0], position)
            .unwrap()
            .outcome;
        self.state = CellState::decode_bound(&batch.state, CELL, position, self.limits).unwrap();
        assert_eq!(self.state.physical_history_version(), 3);
        assert_eq!(batch.state[8], 3);
        assert_eq!(batch.results[0][8], 2);
        let db = self.state.database(DATABASE.database_id).unwrap();
        self.labels.push(format!(
            "{}\t{}\t{label}\t{}\t{}\t{}\t{}\t{}\t{}",
            self.commands.len(),
            position.index,
            outcome_name(&outcome),
            db.revision,
            db.base_revision,
            db.cold,
            batch.state.len(),
            hex(&payload_digest(&batch.state))
        ));
        self.commands.push(bytes);
        self.outcomes.push(batch.results[0].clone());
        self.prefixes.push(batch.state);
        outcome
    }

    fn try_commit(
        &mut self,
        label: &str,
        id: u128,
        result_len: usize,
        page: Option<u8>,
    ) -> (Command, Outcome) {
        let db = self.state.database(DATABASE.database_id).unwrap();
        let ticket = CommitTicket {
            cell_id: CELL.cell_id,
            cell_incarnation: CELL.cell_incarnation,
            database_id: DATABASE.database_id,
            incarnation: DATABASE.incarnation,
            placement_generation: DATABASE.placement_generation,
            owner_epoch: db.owner.epoch,
            worker_instance_id: db.owner.worker_instance_id,
            auth_epoch: db.auth_epoch,
            request_id: id,
            predecessor: self.image.head().revision,
            payload_digest: [0; 32],
        };
        let batch = if let Some(value) = page {
            let mut bytes = vec![0; physical::PAGE_SIZE];
            bytes[..16].copy_from_slice(b"SQLite format 3\0");
            bytes[16..18].copy_from_slice(&[0x10, 0]);
            bytes[100] = value;
            physical::PageBatch {
                size_after: 1,
                pages: vec![physical::PageChange { number: 1, bytes }],
            }
        } else {
            physical::PageBatch {
                size_after: (self.image.bytes().len() / physical::PAGE_SIZE) as u32,
                pages: Vec::new(),
            }
        };
        let prepared = self.image.prepare(ticket, batch).unwrap();
        let mut result = vec![(id - 100) as u8; result_len];
        if result.len() >= 3 {
            result[..3].copy_from_slice(&[(id - 100) as u8, 0, 255]);
        }
        let command = self.command(CommandKind::Commit {
            write: prepared.write().clone(),
            request_fingerprint: [(id - 100) as u8; 32],
            result,
        });
        let outcome = self.apply_command(label, command.clone());
        if let Outcome::Accepted(commit) = &outcome {
            self.image = prepared.accepted(&commit.receipt).unwrap();
        }
        (command, outcome)
    }

    fn cancel(&mut self, label: &str, id: u128) -> Outcome {
        self.apply(
            label,
            CommandKind::ResolveOrCancel {
                request_id: id,
                request_fingerprint: [id as u8; 32],
            },
        )
    }

    fn save(&self, directory: &Path, profile: &str) {
        let stem = directory.join(format!("retention-v3-{profile}"));
        rows(&stem, "commands.hex", &self.commands);
        rows(&stem, "outcomes.hex", &self.outcomes);
        rows(&stem, "prefixes.hex", &self.prefixes);
        rows(
            &stem,
            "state-sha256.hex",
            &self.prefixes[1..]
                .iter()
                .map(|state| payload_digest(state).to_vec())
                .collect::<Vec<_>>(),
        );
        fs::write(
            stem.with_extension("labels.tsv"),
            format!("{}\n", self.labels.join("\n")),
        )
        .unwrap();
        fs::write(
            stem.with_extension("image.hex"),
            format!("{}\n", hex(self.image.bytes())),
        )
        .unwrap();
        println!(
            "{profile}: {} commands, {} prefixes, final index={}, revision={}, bytes={}",
            self.commands.len(),
            self.prefixes.len(),
            self.state.position().index,
            self.state.database(DATABASE.database_id).unwrap().revision,
            self.state.encode().len()
        );
    }
}

fn lifecycle(directory: &Path) {
    let mut trace = Trace::new(limits());
    let (first, first_result) = trace.try_commit("commit_1_page", 101, 3, Some(17));
    let Outcome::Accepted(first_receipt) = first_result else {
        panic!("first commit failed")
    };
    assert_eq!(trace.cancel("cancel_900", 900), Outcome::Cancelled);
    for id in 102..=108 {
        let (_, result) = trace.try_commit("commit_empty_delta", id, 3, None);
        assert!(matches!(result, Outcome::Accepted(_)));
    }
    assert_eq!(trace.state.database(9).unwrap().base_revision, 8);
    assert_eq!(
        trace.apply_command("duplicate_1_after_fold", first),
        Outcome::AlreadyCommitted(first_receipt.clone())
    );
    assert_eq!(
        trace.cancel("cancel_900_again", 900),
        Outcome::AlreadyCancelled
    );
    assert_eq!(
        trace.apply(
            "cancel_committed_1",
            CommandKind::ResolveOrCancel {
                request_id: 101,
                request_fingerprint: [1; 32],
            }
        ),
        Outcome::AlreadyCommitted(first_receipt.clone())
    );
    assert!(matches!(
        trace
            .try_commit("commit_9_replace_page", 109, 900, Some(73))
            .1,
        Outcome::Accepted(_)
    ));
    assert!(matches!(
        trace.try_commit("commit_10_empty_delta", 110, 3, None).1,
        Outcome::Accepted(_)
    ));
    assert_eq!(
        trace.cancel("cancel_901_fill_window", 901),
        Outcome::Cancelled
    );
    assert_eq!(
        trace.try_commit("request_capacity_commit", 111, 3, None).1,
        Outcome::Rejected(Reject::Capacity)
    );
    assert_eq!(
        trace.cancel("request_capacity_cancel", 902),
        Outcome::Rejected(Reject::Capacity)
    );
    assert_eq!(
        trace.apply(
            "duplicate_wrong_fingerprint",
            CommandKind::ResolveOrCancel {
                request_id: 101,
                request_fingerprint: [255; 32],
            }
        ),
        Outcome::Rejected(Reject::Conflict)
    );
    let db = trace.state.database(9).unwrap();
    assert_eq!(
        trace.apply(
            "suspend",
            CommandKind::SuspendDb {
                operation_id: 10,
                token: db.owner,
                expected_revision: db.revision,
            }
        ),
        Outcome::Administrative(AdminResult::Suspended(2))
    );
    let (archive, bytes) = trace
        .state
        .archive_snapshot(DATABASE, trace.admission())
        .unwrap();
    assert_eq!(bytes[8], 3);
    fs::write(
        directory.join("retention-v3-lifecycle.archive.hex"),
        format!("{}\n", hex(&bytes)),
    )
    .unwrap();
    let owner = trace.state.database(9).unwrap().owner;
    assert_eq!(
        trace.apply(
            "publish_archive",
            CommandKind::PublishArchive {
                operation_id: 11,
                expected_owner: owner,
                expected_previous: None,
                archive,
            }
        ),
        Outcome::Administrative(AdminResult::ArchivePublished(archive.object_digest))
    );
    assert_eq!(
        trace.apply(
            "prune_archive",
            CommandKind::PruneArchive {
                operation_id: 12,
                expected_owner: owner,
                archive_digest: archive.object_digest,
            }
        ),
        Outcome::Administrative(AdminResult::ArchivePruned(archive.object_digest))
    );
    assert_eq!(
        trace.cancel("known_cancel_while_cold", 900),
        Outcome::AlreadyCancelled
    );
    assert_eq!(
        trace.try_commit("new_commit_while_cold", 111, 3, None).1,
        Outcome::Rejected(Reject::Fenced)
    );
    assert_eq!(
        trace.apply(
            "auth_fence_after_archive",
            CommandKind::AuthFence {
                operation_id: 13,
                expected_epoch: 1,
                policy: vec![
                    Grant {
                        principal_id: 12,
                        permissions: READ | WRITE | ADMIN
                    },
                    Grant {
                        principal_id: 22,
                        permissions: READ
                    },
                ],
            }
        ),
        Outcome::Administrative(AdminResult::AuthChanged(2))
    );
    let mut stale = trace.command(CommandKind::ResolveOrCancel {
        request_id: 900,
        request_fingerprint: [900u128 as u8; 32],
    });
    stale.admission.auth_epoch = 1;
    assert_eq!(
        trace.apply_command("stale_auth_cancellation", stale),
        Outcome::Rejected(Reject::Unauthorized)
    );
    assert_eq!(
        trace.apply(
            "current_auth_cold_committed_result",
            CommandKind::ResolveOrCancel {
                request_id: 101,
                request_fingerprint: [1; 32],
            }
        ),
        Outcome::AlreadyCommitted(first_receipt.clone())
    );
    assert_eq!(
        trace
            .try_commit("cold_current_auth_duplicate_commit", 101, 8, None)
            .1,
        Outcome::AlreadyCommitted(first_receipt.clone())
    );
    assert_eq!(
        trace.apply(
            "hydrate_old_archive_keeps_current_auth",
            CommandKind::HydrateArchive {
                operation_id: 14,
                archive: bytes,
            }
        ),
        Outcome::Administrative(AdminResult::ArchiveHydrated(archive.object_digest))
    );
    assert_eq!(trace.state.database(9).unwrap().auth_epoch, 2);
    let claim = trace.apply(
        "claim_recovered_owner",
        CommandKind::ClaimOwner {
            operation_id: 15,
            expected_epoch: 2,
            worker_instance_id: 27,
        },
    );
    let Outcome::Administrative(AdminResult::OwnerClaimed(owner)) = claim else {
        panic!("claim after hydrate failed")
    };
    let db = trace.state.database(9).unwrap();
    assert_eq!(
        trace.apply(
            "activate_recovered_owner",
            CommandKind::ActivateOwner {
                operation_id: 16,
                token: owner,
                restored_revision: db.revision,
                restored_hash: db.commit_hash,
            }
        ),
        Outcome::Administrative(AdminResult::OwnerActivated)
    );
    assert_eq!(
        trace
            .try_commit("duplicate_under_recovered_owner", 101, 8, None)
            .1,
        Outcome::AlreadyCommitted(first_receipt)
    );
    assert_eq!(
        trace
            .try_commit("request_capacity_after_hydrate", 111, 3, None)
            .1,
        Outcome::Rejected(Reject::Capacity)
    );
    assert_eq!(
        trace.cancel("cancel_capacity_after_hydrate", 902),
        Outcome::Rejected(Reject::Capacity)
    );
    let source = trace
        .state
        .recovery_source(DATABASE, trace.admission())
        .unwrap();
    let base = source.base.unwrap();
    let mut recovered = physical::Image::from_snapshot(
        physical::Snapshot {
            head: physical::Head {
                identity: physical::EngineIdentity {
                    cell: CELL,
                    database: DATABASE,
                },
                revision: base.revision,
                image_digest: base.image_digest,
            },
            bytes: base.bytes.clone(),
        },
        trace.image.limits(),
    )
    .unwrap();
    for transaction in source.tail {
        recovered = recovered
            .replay(&transaction.commit.receipt, &transaction.physical_changes)
            .unwrap();
    }
    assert_eq!(recovered, trace.image);
    trace.save(directory, "lifecycle");
}

fn capacity(directory: &Path) {
    let mut bounded = limits();
    bounded.max_state_bytes = 8192;
    bounded.max_requests_per_database = 64;
    let mut trace = Trace::new(bounded);
    let (first, first_result) = trace.try_commit("commit_result_1300", 101, 1300, None);
    let Outcome::Accepted(first_receipt) = first_result else {
        panic!("first bounded commit failed")
    };
    for id in 102..=103 {
        assert!(matches!(
            trace.try_commit("commit_result_1300", id, 1300, None).1,
            Outcome::Accepted(_)
        ));
    }
    assert_eq!(
        trace
            .try_commit("state_byte_capacity_commit", 104, 1300, None)
            .1,
        Outcome::Rejected(Reject::Capacity)
    );
    let mut rejected = false;
    for id in 900..964 {
        let result = trace.cancel("fill_bytes_with_cancel", id);
        if result == Outcome::Rejected(Reject::Capacity) {
            rejected = true;
            break;
        }
        assert_eq!(result, Outcome::Cancelled);
    }
    assert!(rejected);
    assert_eq!(
        trace.apply_command("full_state_duplicate_preserves_result", first),
        Outcome::AlreadyCommitted(first_receipt)
    );
    assert_eq!(
        trace.cancel("full_state_existing_cancel", 900),
        Outcome::AlreadyCancelled
    );
    assert_eq!(
        trace
            .try_commit("full_state_repeated_commit_capacity", 104, 3, None)
            .1,
        Outcome::Rejected(Reject::Capacity)
    );
    trace.save(directory, "capacity");
}

fn main() {
    let directory = std::env::args_os()
        .nth(1)
        .expect("output directory argument");
    let directory = Path::new(&directory);
    fs::create_dir_all(directory).unwrap();
    lifecycle(directory);
    capacity(directory);
}
